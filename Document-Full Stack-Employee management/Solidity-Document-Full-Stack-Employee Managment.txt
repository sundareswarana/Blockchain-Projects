pragma solidity ^0.8.20;

contract EmployeeManagement {

    struct Employee {
        string employeeName;
        string designation;
        string salary;
        string location;

        // IPFS CID/URL of employee image
        string photo;

        // IPFS CID/URL of employee PDF document
        string document;

        bool exists;
    }

    mapping(string => Employee) public employees;

    function registerEmployee(
        string memory _employeeId,
        string memory _employeeName,
        string memory _designation,
        string memory _salary,
        string memory _location,
        string memory _photo,
        string memory _document
    ) public {

        require(
            !employees[_employeeId].exists,
            "Employee already exists"
        );

        employees[_employeeId] = Employee(
            _employeeName,
            _designation,
            _salary,
            _location,
            _photo,
            _document,
            true
        );
    }

    function getEmployee(
        string memory _employeeId
    )
        public
        view
        returns (
            string memory,
            string memory,
            string memory,
            string memory,
            string memory,
            string memory
        )
    {
        Employee memory e = employees[_employeeId];

        return (
            e.employeeName,
            e.designation,
            e.salary,
            e.location,
            e.photo,
            e.document
        );
    }
}
